import streamlit as st
from PIL import Image

from src.retrieval import load_retrieval_artifacts


st.set_page_config(
    page_title="Explainable Medical VLM",
    page_icon="🩺",
    layout="wide",
)


@st.cache_resource
def load_artifacts():
    return load_retrieval_artifacts()


st.title("🩺 Explainable Medical VLM")

st.markdown(
    """
    **MedGemma + Visual Prototype Retrieval**

    Upload a medical image to explore VLM-based analysis
    and visually similar prototype cases.
    """
)

try:
    artifacts = load_artifacts()

    st.success("Retrieval artifacts loaded successfully.")

except Exception as e:
    st.error("Could not load retrieval artifacts.")
    st.exception(e)
    st.stop()


uploaded_file = st.file_uploader(
    "Upload a medical image",
    type=["png", "jpg", "jpeg"],
)


if uploaded_file is not None:

    image = Image.open(uploaded_file).convert("RGB")

    st.subheader("Uploaded Image")

    col1, col2 = st.columns([1, 1])

    with col1:
        st.image(
            image,
            caption="Input medical image",
            use_container_width=True,
        )

    with col2:
        st.subheader("Model Analysis")

        question = st.text_input(
            "Ask a question about this image",
            value="What findings are visible in this image?",
        )

        if st.button("Analyze", type="primary"):

            st.info(
                "MedGemma inference will be connected here."
            )

            st.write("Question:", question)

    st.divider()

    st.subheader("Visual Prototype Retrieval")

    st.info(
        "Image-to-embedding retrieval will be connected "
        "after the original embedding pipeline is verified."
    )

    st.write(
        f"Available visual embeddings: "
        f"{len(artifacts['visual_embeddings']):,}"
    )

    st.write(
        f"Available prototypes: "
        f"{len(artifacts['cluster_centroids']):,}"
    )
